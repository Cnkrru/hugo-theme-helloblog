"\t" " " -}} {{- $trimmed := trim $raw " \t\r" -}} {{- if $trimmed -}} {{- $indent := sub (len $raw) (len (replaceRE "^[ \t]+" "" $raw)) -}} {{- if eq $depth 0 -}} {{- $stack = $stack | append $indent -}} {{- $html = printf "%s
    " $html -}} {{- $depth = 1 -}} {{- else if gt $indent (index $stack (sub $depth 1)) -}} {{- $stack = $stack | append $indent -}} {{- $html = printf "%s
      " $html -}} {{- $depth = add $depth 1 -}} {{- else -}} {{- range seq 1 $depth -}} {{- if and (gt $depth 1) (lt $indent (index $stack (sub $depth 1))) -}} {{- $html = printf "%s
    " $html -}} {{- $stack = first (sub (len $stack) 1) $stack -}} {{- $depth = sub $depth 1 -}} {{- end -}} {{- end -}} {{- $html = printf "%s" $html -}} {{- end -}} {{- $fields := partial "components/splitRow.html" (replaceRE "^[-*+][ \t]+" "" $trimmed) -}} {{- $name := "" -}}{{- $note := "" -}} {{- if ge (len $fields) 1 -}}{{- $name = index $fields 0 -}}{{- end -}} {{- if ge (len $fields) 2 -}}{{- $note = index $fields 1 -}}{{- end -}} {{- $isDir := hasSuffix $name "/" -}} {{- $label := cond $isDir (strings.TrimSuffix "/" $name) $name -}} {{- $icon := cond $isDir $dirIcon $fileIcon -}} {{- $class := cond $isDir "is-dir" "is-file" -}} {{- $html = printf `%s
  • %s%s` $html $class $icon (htmlEscape $label) -}} {{- if $note -}} {{- $html = printf `%s%s` $html (htmlEscape $note) -}} {{- end -}} {{- end -}} {{- end -}} {{- range seq 1 $depth -}} {{- $html = printf "%s
" $html -}} {{- end -}}
