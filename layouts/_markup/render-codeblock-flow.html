{{/* 简易流程：每行 节点 | 可选说明，节点之间自动向下箭头 */}} {{- $arrow := `` -}} {{- $lines := slice -}} {{- range split .Inner "\n" -}} {{- $line := trim . " \t\r" -}} {{- if $line -}}{{- $lines = $lines | append $line -}}{{- end -}} {{- end -}} {{- $total := len $lines -}} {{- $items := "" -}} {{- range $i, $line := $lines -}} {{- $fields := partial "components/splitRow.html" $line -}} {{- $text := "" -}}{{- $sub := "" -}} {{- if ge (len $fields) 1 -}}{{- $text = index $fields 0 -}}{{- end -}} {{- if ge (len $fields) 2 -}}{{- $sub = index $fields 1 -}}{{- end -}} {{- $item := printf `
%s
` (htmlEscape $text) -}} {{- if $sub -}}{{- $item = printf `%s
%s
` $item (htmlEscape $sub) -}}{{- end -}} {{- $item = printf `%s
` $item -}} {{- if lt (add $i 1) $total -}} {{- $item = printf `%s
%s
` $item $arrow -}} {{- end -}} {{- $items = printf "%s%s" $items $item -}} {{- end -}}
{{ $items | safeHTML }}
